cmake_minimum_required(VERSION 3.22)
project(NinjaLegacyParserProbe LANGUAGES C)
get_filename_component(NR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(NR_SPINE_21 OFF CACHE BOOL "Compare official Spine 2.1.25 runtime")
if(NR_SPINE_21)
  set(LEGACY "${NR_ROOT}/build/deps/spine-2.1.25")
  set(EXPECTED_REV "142e770e5b4b42395a3a0a68ff3c8a21ba2774eb")
  set(SPINE_SOURCE "${LEGACY}/spine-c/src/spine")
  set(SPINE_INCLUDE "${LEGACY}/spine-c/include")
else()
  set(LEGACY "${NR_ROOT}/build/deps/cocos2d-x-3.5")
  set(EXPECTED_REV "f7e11a2467e4508738ad9cf4b3992da05f9d1e7c")
  set(SPINE_SOURCE "${LEGACY}/cocos/editor-support/spine")
  set(SPINE_INCLUDE "${LEGACY}/cocos/editor-support")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LEGACY}" rev-parse HEAD
  OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT REV STREQUAL EXPECTED_REV)
  message(FATAL_ERROR "Legacy reference revision mismatch")
endif()
# Uses unchanged reference C parser for internal comparison, no engine/graphics backend.
file(GLOB SPINE_C "${SPINE_SOURCE}/*.c")
add_executable(NinjaLegacyParserProbe parser_probe.c ${SPINE_C})
if(WIN32)
  target_link_libraries(NinjaLegacyParserProbe PRIVATE dbghelp)
endif()
target_include_directories(NinjaLegacyParserProbe PRIVATE "${SPINE_INCLUDE}")
if(NOT MSVC)
  target_link_libraries(NinjaLegacyParserProbe PRIVATE m)
endif()
enable_testing()
add_test(NAME original_spine_2_1_parse COMMAND NinjaLegacyParserProbe
  "${NR_ROOT}/01_conteudo_apk/assets/res/sprite/spirte_102101.json"
  "${NR_ROOT}/01_conteudo_apk/assets/res/sprite/spirte_102101.atlas")
set_tests_properties(original_spine_2_1_parse PROPERTIES TIMEOUT 20)
