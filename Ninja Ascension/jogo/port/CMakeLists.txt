cmake_minimum_required(VERSION 3.22...4.1)
project(NinjaResourceProbe LANGUAGES C CXX)
# Diagnostic target only. Production gate has NOT passed.
set(APP_NAME NinjaResourceProbe)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This probe requires a native Windows x64 toolchain")
endif()
get_filename_component(NR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(AX_ROOT "${NR_ROOT}/build/deps/axmol" CACHE PATH "Pinned Axmol checkout")
find_package(Git)
set(AX_REV "")
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AX_ROOT}" rev-parse HEAD
    OUTPUT_VARIABLE AX_REV RESULT_VARIABLE AX_REV_RESULT
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT AX_REV_RESULT EQUAL 0)
    set(AX_REV "")
  endif()
endif()
if(AX_REV)
  if(NOT AX_REV STREQUAL "6c0dce6db4660feafe29dabb1ab3506ee76b0931")
    message(FATAL_ERROR "Expected Axmol v2.11.5 pinned revision; found ${AX_REV}")
  endif()
else()
  # Portable packages ship the engine sources without .git history.
  # (core/axmolver.h is regenerated by the engine on every configure, so it
  # cannot prove anything.) Require the version template and the release
  # notes of the pinned 2.11.5 tree instead.
  file(STRINGS "${AX_ROOT}/core/axmolver.h.in" AX_VER_LINES
       REGEX "#define AX_VERSION_(MAJOR|MINOR|PATCH) ")
  string(REGEX MATCH "AX_VERSION_MAJOR ([0-9]+)" _m "${AX_VER_LINES}")
  set(_ax_major "${CMAKE_MATCH_1}")
  string(REGEX MATCH "AX_VERSION_MINOR ([0-9]+)" _m "${AX_VER_LINES}")
  set(_ax_minor "${CMAKE_MATCH_1}")
  string(REGEX MATCH "AX_VERSION_PATCH ([0-9]+)" _m "${AX_VER_LINES}")
  set(_ax_patch "${CMAKE_MATCH_1}")
  file(STRINGS "${AX_ROOT}/CHANGELOG.md" _ax_notes LIMIT_COUNT 1)
  if(NOT "${_ax_major}.${_ax_minor}.${_ax_patch}" STREQUAL "2.11.5" OR NOT _ax_notes MATCHES "^# axmol-2\\.11\\.5 ")
    message(FATAL_ERROR "Expected Axmol v2.11.5; found ${_ax_major}.${_ax_minor}.${_ax_patch} / ${_ax_notes}")
  endif()
  message(STATUS "Axmol 2.11.5 confirmed from axmolver.h.in and CHANGELOG (no git history in this package)")
endif()
set(_AX_ROOT "${AX_ROOT}")
set(ENV{AX_ROOT} "${AX_ROOT}")
set(_AX_LUA_PROJ TRUE)
set(AX_EXT_HINT OFF CACHE BOOL "Only enable extensions used by the resource probe" FORCE)
set(AX_ENABLE_EXT_GUI ON CACHE BOOL "" FORCE)
set(AX_ENABLE_MSEDGE_WEBVIEW2 OFF CACHE BOOL "No embedded browser in offline resource probe" FORCE)
set(AX_USE_LUAJIT ON CACHE BOOL "Use Lua 5.1 compatible runtime" FORCE)
set(AX_ENABLE_EXT_LUA ON CACHE BOOL "" FORCE)
set(AX_ENABLE_EXT_COCOSTUDIO ON CACHE BOOL "" FORCE)
set(AX_ENABLE_AUDIO ON CACHE BOOL "" FORCE)
set(AX_ENABLE_EXT_SPINE OFF CACHE BOOL "4.2 cannot load original 2.1.27" FORCE)
foreach(extension ASSETMANAGER DRAGONBONES FAIRYGUI IMGUI INSPECTOR SDFGEN JSONDEFAULT PHYSICS_NODE PARTICLE3D)
  set(AX_ENABLE_EXT_${extension} OFF CACHE BOOL "Unused by resource diagnostic" FORCE)
endforeach()
set(AX_ENABLE_EXT_IMGUI ON CACHE BOOL "Inspector visual" FORCE)
set(AX_ENABLE_EXT_INSPECTOR ON CACHE BOOL "Inspector visual" FORCE)
# Generated Lua extension bindings reference the 3D particle API in this revision.
# Keep its implementation linked even though this diagnostic does not instantiate it.
set(AX_ENABLE_EXT_PARTICLE3D ON CACHE BOOL "Required by pinned generated Lua bindings" FORCE)
set(AX_ENABLE_EXT_ASSETMANAGER ON CACHE BOOL "Required by pinned manual Lua bindings" FORCE)
list(APPEND CMAKE_MODULE_PATH "${AX_ROOT}/cmake/Modules")
include(AXBuildSet)
add_subdirectory("${AX_ROOT}/core" "${ENGINE_BINARY_PATH}/axmol/core")
add_library(nr_layout STATIC native/LayoutOverrides.cpp)
target_compile_features(nr_layout PUBLIC cxx_std_20)
target_include_directories(nr_layout PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native" PRIVATE "${AX_ROOT}/3rdparty")
add_executable(LayoutOverridesTests tests/LayoutOverridesTests.cpp)
target_link_libraries(LayoutOverridesTests PRIVATE nr_layout)
set(SPINE21 "${NR_ROOT}/build/deps/spine-2.1.25")
set(SPINE_REV "")
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SPINE21}" rev-parse HEAD
    OUTPUT_VARIABLE SPINE_REV RESULT_VARIABLE SPINE_REV_RESULT
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT SPINE_REV_RESULT EQUAL 0)
    set(SPINE_REV "")
  endif()
endif()
if(SPINE_REV)
  if(NOT SPINE_REV STREQUAL "142e770e5b4b42395a3a0a68ff3c8a21ba2774eb")
    message(FATAL_ERROR "Spine 2.1.25 reference revision mismatch")
  endif()
else()
  # Portable package without .git: require the exact 2.1.25 spine-c layout
  # (26 runtime sources, public header) that the native adapter was built for.
  file(GLOB _spine_c_files "${SPINE21}/spine-c/src/spine/*.c")
  list(LENGTH _spine_c_files _spine_c_count)
  if(NOT _spine_c_count EQUAL 26 OR NOT EXISTS "${SPINE21}/spine-c/include/spine/spine.h")
    message(FATAL_ERROR "Spine 2.1.25 sources incomplete (${_spine_c_count} .c files)")
  endif()
  message(STATUS "Spine 2.1.25 sources confirmed (no git history in this package)")
endif()
file(GLOB SPINE21_C CONFIGURE_DEPENDS "${SPINE21}/spine-c/src/spine/*.c")
add_library(nr_spine21 STATIC ${SPINE21_C})
target_include_directories(nr_spine21 PUBLIC "${SPINE21}/spine-c/include")
add_executable(${APP_NAME} native/ResourceProbe.cpp native/LayoutEditor.cpp native/LegacyCharacter.cpp native/LocalBattle.cpp native/protocol/NinjaProtoNatives.cpp)
configure_file(native/NinjaAscension.rc.in ${CMAKE_CURRENT_BINARY_DIR}/NinjaAscension.rc @ONLY)
target_sources(${APP_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/NinjaAscension.rc)
target_compile_features(${APP_NAME} PRIVATE cxx_std_20)
target_link_libraries(${APP_NAME} axlua cocostudio nr_layout nr_spine21 zlib)
# Third-party import libs still pull the release MSVCRT (LNK4098 under the debug
# runtime); skip exactly that defaultlib and let the link prove nothing needs it.
# Release keeps a PDB (kept on the build machine, not shipped) so crash
# reports from players' PCs can be resolved to function and line.
target_compile_options(${APP_NAME} PRIVATE "$<$<CONFIG:Release>:/Zi>")
target_link_options(${APP_NAME} PRIVATE "$<$<CONFIG:Release>:/DEBUG:FULL;/OPT:REF;/OPT:ICF>")
# Release links against MSVCRT itself, so the exclusion is Debug-only.
target_link_options(${APP_NAME} PRIVATE "$<$<CONFIG:Debug>:/NODEFAULTLIB:MSVCRT>")
set(NR_REFERENCE_ROOT "${NR_ROOT}" CACHE PATH "Original workspace path for runtime resource reads")
target_compile_definitions(${APP_NAME} PRIVATE NR_ROOT="${NR_REFERENCE_ROOT}")
ax_setup_app_config(${APP_NAME} CONSOLE)
ax_sync_target_dlls(${APP_NAME})
ax_setup_app_props(${APP_NAME})
